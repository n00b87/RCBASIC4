#header Sub SetSpriteCanvasBlendMode( c_num, blend_mode )

#li BLEND_MODE_ADD

#li BLEND_MODE_MIN
#li BLEND_MODE_MAX
#li BLEND_MODE_MIN_FACTOR
#li BLEND_MODE_MAX_FACTOR
#li BLEND_MODE_MIN_ALPHA
#li BLEND_MODE_MAX_ALPHA
